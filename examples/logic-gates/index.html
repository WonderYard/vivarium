Logic Gates

Source

Two clocks feed each gate: OR, AND and XOR. A gate is a single cell that counts the pulses pointing at it. 

const { TOP, RIGHT, BOTTOM, LEFT } = neighbor

          
// Only the pulsing state points somewhere.

          
const part = (name, outputs, count) => {

          
  const idle = element(name, "#334155")

          
  const on = element(`${name} on`, "#facc15", [], outputs)

          
  idle.to(on).pointed(count)

          
  on.to(idle).pointed(not(count))

          
}

          

          
part("right", [RIGHT], between(1, 4)) // and left, up, down

          
part("splitter", [BOTTOM, RIGHT], between(1, 4))

          

          
// A gate is one cell: its count is its truth table.

          
part("or", [RIGHT], between(1, 4))

          
part("and", [RIGHT], [2])

          
part("xor", [RIGHT], [1])
